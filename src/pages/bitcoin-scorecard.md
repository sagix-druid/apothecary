    async function updateBitcoinData() {
      const loadingElements = document.querySelectorAll('.loading');
      loadingElements.forEach(el => el.style.display = 'inline');
      try {
        const response = await fetch('https://api.coingecko.com/api/v3/coins/bitcoin?localization=false&tickers=false&market_data=true&community_data=false&developer_data=false&sparkline=false');
        const data = await response.json();
        // Update price
        const price = data.market_data.current_price.usd;
        document.getElementById('btc-price').textContent = '$' + price.toLocaleString();
        // Update 7D change
        const change7d = data.market_data.price_change_percentage_7d;
        const changeElement = document.getElementById('btc-change');
        changeElement.textContent = (change7d > 0 ? '+' : '') + change7d.toFixed(1) + '% (7D)';
        changeElement.style.color = change7d > 0 ? '#22c55e' : '#ef4444';
        // Update market cap
        const marketCap = data.market_data.market_cap.usd;
        const marketCapFormatted = marketCap > 1e12 ?
          '$' + (marketCap / 1e12).toFixed(1) + 'T' :
          '$' + (marketCap / 1e9).toFixed(0) + 'B';
        document.getElementById('btc-marketcap').textContent = marketCapFormatted;
        // Fetch Lightning Network capacity
        try {
          const lightningResponse = await fetch('https://mempool.space/api/v1/lightning/statistics/latest');
          const lightningData = await lightningResponse.json();
          const capacityBTC = lightningData.total_capacity / 100000000;
          const capacityFormatted = capacityBTC > 1000 ? 
            (capacityBTC / 1000).toFixed(0) + 'K BTC' : 
            capacityBTC.toFixed(0) + ' BTC';
          document.getElementById('btc-lightning').textContent = capacityFormatted + ' Capacity';
        } catch (lightningError) {
          document.getElementById('btc-lightning').textContent = 'Data unavailable';
        }
        loadingElements.forEach(el => el.style.display = 'none');
      } catch (error) {
        console.error('Could not fetch live Bitcoin data:', error);
        document.getElementById('btc-price').textContent = 'Price data unavailable';
        document.getElementById('btc-change').textContent = 'Change data unavailable';
        document.getElementById('btc-marketcap').textContent = 'Market cap unavailable';
        document.getElementById('btc-lightning').textContent = 'Lightning data unavailable';
      }
    }
    document.addEventListener('DOMContentLoaded', updateBitcoinData);
  </script>
