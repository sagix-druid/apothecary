Bittensor (TAO) is revolutionizing the intersection of blockchain and artificial intelligence by establishing a decentralized marketplace for machine intelligence. Launched in 2021, it draws inspiration from Bitcoin's tokenomics, featuring a strict 21 million TAO hard cap and deflationary halvings—the first set for December 2025—to ensure scarcity and value. With over 128 specialized subnets powering tasks from AI model training to protein folding, Bittensor transforms computational resources into economically valuable outputs, backed by over $350 million from top investors like Polychain Capital and Digital Currency Group. This infrastructure not only offers staking yields of 14-20% APY but also positions TAO as a key player in the future of decentralized AI.

<h2>About Bittensor</h2>
<p>Bittensor, launched in 2021 by Jacob Steeves, creates a decentralized marketplace for machine intelligence where anyone can contribute computing power, AI models, or data and earn TAO tokens proportionally. The network implements a Bitcoin-inspired fixed supply of 21 million TAO with deflationary halvings every ~4 years, but produces economically valuable AI outputs instead of purposeless hashes. Operating 128+ specialized subnets for tasks ranging from text generation to protein folding, Bittensor has attracted $350M+ in institutional backing from Polychain Capital, Digital Currency Group, and dao5. With 74.9% of circulating supply staked and the first halving event approaching December 2025, TAO combines cryptocurrency scarcity with AI mega-trend exposure through decentralized infrastructure positioning.</p>

  function generateSparklineSVG(prices, width, height, strokeColor) {
    if (!prices || prices.length < 2) {
      return '<svg width="' + width + '" height="' + height + '"></svg>';
    }
    var minPrice = Math.min.apply(Math, prices);
    var maxPrice = Math.max.apply(Math, prices);
    var priceRange = maxPrice - minPrice || 1;
    var points = prices.map(function(price, i) {
      var x = (i / (prices.length - 1)) * width;
      var y = height - ((price - minPrice) / priceRange) * height;
      return x + ',' + y;
    }).join(' ');
    return '<svg width="' + width + '" height="' + height + '" viewBox="0 0 ' + width + ' ' + height + '" xmlns="http://www.w3.org/2000/svg"><polyline fill="none" stroke="' + strokeColor + '" stroke-width="2" points="' + points + '" /></svg>';
  }
  async function updateBittensorData() {
    try {
      var response = await fetch('https://api.coingecko.com/api/v3/coins/bittensor?localization=false&tickers=false&market_data=true&community_data=false&developer_data=false&sparkline=true');
      var data = await response.json();
      if (data && data.market_data) {
        // Update price
        var price = data.market_data.current_price.usd;
        document.getElementById('tao-price').textContent = '$' + price.toLocaleString();
        // Update 7D change
        var change7d = data.market_data.price_change_percentage_7d;
        var changeElement = document.getElementById('tao-change');
        if (change7d !== undefined && change7d !== null) {
          changeElement.textContent = (change7d > 0 ? '+' : '') + change7d.toFixed(1) + '% (7D)';
          changeElement.style.color = change7d > 0 ? '#22c55e' : '#ef4444';
        } else {
          changeElement.textContent = 'Change unavailable';
        }
        // Update market cap
        var marketCap = data.market_data.market_cap.usd;
        var marketCapFormatted = marketCap > 1e12 ? '$' + (marketCap / 1e12).toFixed(1) + 'T' : '$' + (marketCap / 1e9).toFixed(1) + 'B';
        document.getElementById('tao-marketcap').textContent = marketCapFormatted;
        // Generate sparkline
        if (data.market_data.sparkline_in_7d && data.market_data.sparkline_in_7d.price) {
          var sparklinePrices = data.market_data.sparkline_in_7d.price;
          var sparklineColor = change7d > 0 ? '#22c55e' : '#ef4444';
          var svg = generateSparklineSVG(sparklinePrices, 200, 60, sparklineColor);
          document.getElementById('tao-sparkline').innerHTML = svg;
        } else {
          document.getElementById('tao-sparkline').innerHTML = '<img src="https://www.coingecko.com/coins/22974/sparkline.svg" style="width: 100%; height: 60px; object-fit: cover;" alt="Bittensor 7D Chart">';
        }
      }
    } catch (error) {
      console.error('Error:', error);
      document.getElementById('tao-price').textContent = 'Data unavailable';
      document.getElementById('tao-change').textContent = 'Data unavailable';
    }
  }
  document.addEventListener('DOMContentLoaded', updateBittensorData);
