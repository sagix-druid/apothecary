<h2>About Ethereum</h2>
<p>Ethereum, launched in 2015 by Vitalik Buterin, introduced programmable smart contracts to blockchain technology. As the foundation of decentralized finance (DeFi), it hosts over $80 billion in total value locked across thousands of protocols. The successful transition to Proof of Stake in 2022 reduced energy consumption by 99% while introducing staking rewards. With EIP-1559's fee burning mechanism and growing institutional adoption, Ethereum has evolved into both a utility platform and deflationary store of value.</p>

  function generateSparklineSVG(prices, width, height, strokeColor) {
    if (!prices || prices.length < 2) {
      return '<svg width="' + width + '" height="' + height + '"></svg>';
    }
    var minPrice = Math.min.apply(Math, prices);
    var maxPrice = Math.max.apply(Math, prices);
    var priceRange = maxPrice - minPrice || 1;
    var points = prices.map(function(price, i) {
      var x = (i / (prices.length - 1)) * width;
      var y = height - ((price - minPrice) / priceRange) * height;
      return x + ',' + y;
    }).join(' ');
    return '<svg width="' + width + '" height="' + height + '" viewBox="0 0 ' + width + ' ' + height + '" xmlns="http://www.w3.org/2000/svg"><polyline fill="none" stroke="' + strokeColor + '" stroke-width="2" points="' + points + '" /></svg>';
  }
  async function updateEthereumData() {
    try {
      var response = await fetch('https://api.coingecko.com/api/v3/coins/ethereum?localization=false&tickers=false&market_data=true&community_data=false&developer_data=false&sparkline=true');
      var data = await response.json();
      if (data && data.market_data) {
        // Update price
        var price = data.market_data.current_price.usd;
        document.getElementById('eth-price').textContent = '$' + price.toLocaleString();
        // Update 7D change
        var change7d = data.market_data.price_change_percentage_7d;
        var changeElement = document.getElementById('eth-change');
        if (change7d !== undefined && change7d !== null) {
          changeElement.textContent = (change7d > 0 ? '+' : '') + change7d.toFixed(1) + '% (7D)';
          changeElement.style.color = change7d > 0 ? '#22c55e' : '#ef4444';
        } else {
          changeElement.textContent = 'Change unavailable';
        }
        // Update market cap
        var marketCap = data.market_data.market_cap.usd;
        var marketCapFormatted = marketCap > 1e12 ? '$' + (marketCap / 1e12).toFixed(1) + 'T' : '$' + (marketCap / 1e9).toFixed(0) + 'B';
        document.getElementById('eth-marketcap').textContent = marketCapFormatted;
        // Generate sparkline
        if (data.market_data.sparkline_in_7d && data.market_data.sparkline_in_7d.price) {
          var sparklinePrices = data.market_data.sparkline_in_7d.price;
          var sparklineColor = change7d > 0 ? '#22c55e' : '#ef4444';
          var svg = generateSparklineSVG(sparklinePrices, 200, 60, sparklineColor);
          document.getElementById('eth-sparkline').innerHTML = svg;
        } else {
          document.getElementById('eth-sparkline').innerHTML = '<img src="https://www.coingecko.com/coins/279/sparkline.svg" style="width: 100%; height: 60px; object-fit: cover;" alt="Ethereum 7D Chart">';
        }
      }
      // Fetch TVL
      try {
        var tvlResponse = await fetch('https://api.llama.fi/tvl/ethereum');
        var tvlData = await tvlResponse.json();
        var tvlFormatted = tvlData > 1e12 ? '$' + (tvlData / 1e12).toFixed(1) + 'T' : '$' + (tvlData / 1e9).toFixed(0) + 'B';
        document.getElementById('eth-tvl').textContent = tvlFormatted;
      } catch (e) {
        document.getElementById('eth-tvl').textContent = 'Data unavailable';
      }
    } catch (error) {
      console.error('Error:', error);
      document.getElementById('eth-price').textContent = 'Data unavailable';
      document.getElementById('eth-change').textContent = 'Data unavailable';
    }
  }
  document.addEventListener('DOMContentLoaded', updateEthereumData);
</script>
