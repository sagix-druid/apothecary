<h2>About Spark Protocol</h2>
<p>Spark Protocol, launched as part of the MakerDAO/Sky ecosystem, is an onchain capital allocator managing $3.8B+ across DeFi, CeFi, and real-world assets. The protocol features three core products: SparkLend (lending with governance-defined rates), Spark Savings (yield-bearing stablecoins like sUSDS and sUSDC), and the Spark Liquidity Layer (automated capital deployment across protocols like Aave, Morpho, and BlackRock's BUIDL). The SPK token, launched in June 2025 with a 10B hard cap, enables governance participation and staking for Spark Points rewards. Operating on Ethereum mainnet, Base, and Arbitrum, Spark serves as the liquidity and yield infrastructure powering Sky's vision of bridging traditional and decentralized finance.</p>

  function generateSparklineSVG(prices, width, height, strokeColor) {
    if (!prices || prices.length < 2) {
      return '<svg width="' + width + '" height="' + height + '"></svg>';
    }
    var minPrice = Math.min.apply(Math, prices);
    var maxPrice = Math.max.apply(Math, prices);
    var priceRange = maxPrice - minPrice || 1;
    var points = prices.map(function(price, i) {
      var x = (i / (prices.length - 1)) * width;
      var y = height - ((price - minPrice) / priceRange) * height;
      return x + ',' + y;
    }).join(' ');
    return '<svg width="' + width + '" height="' + height + '" viewBox="0 0 ' + width + ' ' + height + '" xmlns="http://www.w3.org/2000/svg"><polyline fill="none" stroke="' + strokeColor + '" stroke-width="2" points="' + points + '" /></svg>';
  }
  async function updateSparkData() {
    try {
      var response = await fetch('https://api.coingecko.com/api/v3/coins/spark?localization=false&tickers=false&market_data=true&community_data=false&developer_data=false&sparkline=true');
      var data = await response.json();
      if (data && data.market_data) {
        // Update price
        var price = data.market_data.current_price.usd;
        document.getElementById('spk-price').textContent = '$' + price.toLocaleString('en-US', { minimumFractionDigits: 3, maximumFractionDigits: 4 });
        // Update 7D change
        var change7d = data.market_data.price_change_percentage_7d;
        var changeElement = document.getElementById('spk-change');
        if (change7d !== undefined && change7d !== null) {
          changeElement.textContent = (change7d > 0 ? '+' : '') + change7d.toFixed(1) + '% (7D)';
          changeElement.style.color = change7d > 0 ? '#22c55e' : '#ef4444';
        } else {
          changeElement.textContent = 'Change unavailable';
        }
        // Update market cap
        var marketCap = data.market_data.market_cap.usd;
        var marketCapFormatted = marketCap > 1e9 ? '$' + (marketCap / 1e9).toFixed(0) + 'B' : '$' + (marketCap / 1e6).toFixed(0) + 'M';
        document.getElementById('spk-marketcap').textContent = marketCapFormatted;
        // Generate sparkline
        if (data.market_data.sparkline_in_7d && data.market_data.sparkline_in_7d.price) {
          var sparklinePrices = data.market_data.sparkline_in_7d.price;
          var sparklineColor = change7d > 0 ? '#22c55e' : '#ef4444';
          var svg = generateSparklineSVG(sparklinePrices, 200, 60, sparklineColor);
          document.getElementById('spk-sparkline').innerHTML = svg;
        } else {
          // Fallback to static placeholder if sparkline data unavailable
          document.getElementById('spk-sparkline').innerHTML = '<div style="width: 200px; height: 60px; background: #f1f5f9; border-radius: 4px; display: flex; align-items: center; justify-content: center; color: #64748b; font-size: 0.8em;">Chart unavailable</div>';
        }
      }
      // Fetch protocol TVL
      try {
        var tvlResponse = await fetch('https://api.llama.fi/tvl/spark');
        var tvlData = await tvlResponse.json();
        var tvlFormatted = tvlData > 1e12 ? '$' + (tvlData / 1e12).toFixed(1) + 'T' : '$' + (tvlData / 1e9).toFixed(1) + 'B';
        document.getElementById('spk-tvl').textContent = tvlFormatted;
      } catch (e) {
        document.getElementById('spk-tvl').textContent = '~$3.8B';
      }
    } catch (error) {
      console.error('Error:', error);
      document.getElementById('spk-price').textContent = 'Data unavailable';
      document.getElementById('spk-change').textContent = 'Data unavailable';
    }
  }
  document.addEventListener('DOMContentLoaded', updateSparkData);
</script>
