<h2>About Raydium</h2>
<p>Raydium, launched in February 2021, is an automated market maker (AMM) and liquidity provider built on the Solana blockchain. Unlike traditional AMMs, Raydium provides on-chain liquidity to Serum's central limit order book, giving liquidity providers access to the entire order flow. The protocol charges a 0.25% trading fee: 0.22% goes to liquidity providers, while 0.03% rewards RAY stakers. With 555 million RAY tokens hard-capped and fully minted at genesis, the protocol uses 12% of fees for RAY buybacks rather than token burns. Raydium has become Solana's infrastructure backbone, hosting the majority of the network's DeFi trading volume.</p>

  function generateSparklineSVG(prices, width, height, strokeColor) {
    if (!prices || prices.length < 2) {
      return '<svg width="' + width + '" height="' + height + '"></svg>';
    }
    var minPrice = Math.min.apply(Math, prices);
    var maxPrice = Math.max.apply(Math, prices);
    var priceRange = maxPrice - minPrice || 1;
    var points = prices.map(function(price, i) {
      var x = (i / (prices.length - 1)) * width;
      var y = height - ((price - minPrice) / priceRange) * height;
      return x + ',' + y;
    }).join(' ');
    return '<svg width="' + width + '" height="' + height + '" viewBox="0 0 ' + width + ' ' + height + '" xmlns="http://www.w3.org/2000/svg"><polyline fill="none" stroke="' + strokeColor + '" stroke-width="2" points="' + points + '" /></svg>';
  }
  async function updateRaydiumData() {
    try {
      var response = await fetch('https://api.coingecko.com/api/v3/coins/raydium?localization=false&tickers=false&market_data=true&community_data=false&developer_data=false&sparkline=true');
      var data = await response.json();
      if (data && data.market_data) {
        // Update price
        var price = data.market_data.current_price.usd;
        document.getElementById('ray-price').textContent = '$' + price.toLocaleString('en-US', { minimumFractionDigits: 2, maximumFractionDigits: 2 });
        // Update 7D change
        var change7d = data.market_data.price_change_percentage_7d;
        var changeElement = document.getElementById('ray-change');
        if (change7d !== undefined && change7d !== null) {
          changeElement.textContent = (change7d > 0 ? '+' : '') + change7d.toFixed(1) + '% (7D)';
          changeElement.style.color = change7d > 0 ? '#22c55e' : '#ef4444';
        } else {
          changeElement.textContent = 'Change unavailable';
        }
        // Update market cap
        var marketCap = data.market_data.market_cap.usd;
        var marketCapFormatted = marketCap > 1e12 ? '$' + (marketCap / 1e12).toFixed(1) + 'T' : '$' + (marketCap / 1e9).toFixed(2) + 'B';
        document.getElementById('ray-marketcap').textContent = marketCapFormatted;
        // Generate sparkline
        if (data.market_data.sparkline_in_7d && data.market_data.sparkline_in_7d.price) {
          var sparklinePrices = data.market_data.sparkline_in_7d.price;
          var sparklineColor = change7d > 0 ? '#22c55e' : '#ef4444';
          var svg = generateSparklineSVG(sparklinePrices, 200, 60, sparklineColor);
          document.getElementById('ray-sparkline').innerHTML = svg;
        }
      }
      // Fetch TVL from DeFiLlama
      try {
        var tvlResponse = await fetch('https://api.llama.fi/protocol/raydium');
        var tvlData = await tvlResponse.json();
        if (tvlData && tvlData.tvl) {
          var tvl = tvlData.tvl[tvlData.tvl.length - 1].totalLiquidityUSD;
          var tvlFormatted = tvl > 1e12 ? '$' + (tvl / 1e12).toFixed(1) + 'T' : '$' + (tvl / 1e9).toFixed(2) + 'B';
          document.getElementById('ray-tvl').textContent = tvlFormatted;
        }
      } catch (e) {
        document.getElementById('ray-tvl').textContent = 'Data unavailable';
      }
    } catch (error) {
      console.error('Error:', error);
      document.getElementById('ray-price').textContent = 'Data unavailable';
      document.getElementById('ray-change').textContent = 'Data unavailable';
    }
  }
  document.addEventListener('DOMContentLoaded', updateRaydiumData);
