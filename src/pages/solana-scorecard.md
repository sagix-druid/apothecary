This breakthrough enables high-frequency applications like decentralized exchanges, real-time payments, and NFT marketplaces that were previously impossible on slower chains. Powered by SOL for transaction fees and staking, Solana creates a scalable environment where speed and cost efficiency unlock new possibilities for consumer-facing blockchain applications.

<h2>About Solana</h2>
<p>Solana, launched in 2020 by Anatoly Yakovenko and the Solana Labs team, introduced a novel Proof of History (PoH) consensus mechanism that enables exceptional blockchain performance. By creating a verifiable passage of time between events, Solana achieves theoretical throughput of 65,000+ transactions per second with average costs under $0.001. The network hosts over $10B in DeFi TVL across protocols like Jupiter, Raydium, and MarginFi, while also dominating NFT trading volume. SOL tokens are used for transaction fees, staking (with 6-8% yields), and governance, with an inflationary supply schedule that decreases annually toward a long-term 1.5% rate.</p>

  function generateSparklineSVG(prices, width, height, strokeColor) {
    if (!prices || prices.length < 2) {
      return '<svg width="' + width + '" height="' + height + '"></svg>';
    }
    var minPrice = Math.min.apply(Math, prices);
    var maxPrice = Math.max.apply(Math, prices);
    var priceRange = maxPrice - minPrice || 1;
    var points = prices.map(function(price, i) {
      var x = (i / (prices.length - 1)) * width;
      var y = height - ((price - minPrice) / priceRange) * height;
      return x + ',' + y;
    }).join(' ');
    return '<svg width="' + width + '" height="' + height + '" viewBox="0 0 ' + width + ' ' + height + '" xmlns="http://www.w3.org/2000/svg"><polyline fill="none" stroke="' + strokeColor + '" stroke-width="2" points="' + points + '" /></svg>';
  }
  async function updateSolanaData() {
    try {
      var response = await fetch('https://api.coingecko.com/api/v3/coins/solana?localization=false&tickers=false&market_data=true&community_data=false&developer_data=false&sparkline=true');
      var data = await response.json();
      if (data && data.market_data) {
        // Update price
        var price = data.market_data.current_price.usd;
        document.getElementById('sol-price').textContent = '$' + price.toLocaleString();
        // Update 7D change
        var change7d = data.market_data.price_change_percentage_7d;
        var changeElement = document.getElementById('sol-change');
        if (change7d !== undefined && change7d !== null) {
          changeElement.textContent = (change7d > 0 ? '+' : '') + change7d.toFixed(1) + '% (7D)';
          changeElement.style.color = change7d > 0 ? '#22c55e' : '#ef4444';
        } else {
          changeElement.textContent = 'Change unavailable';
        }
        // Update market cap
        var marketCap = data.market_data.market_cap.usd;
        var marketCapFormatted = marketCap > 1e12 ? '$' + (marketCap / 1e12).toFixed(1) + 'T' : '$' + (marketCap / 1e9).toFixed(0) + 'B';
        document.getElementById('sol-marketcap').textContent = marketCapFormatted;
        // Generate sparkline
        if (data.market_data.sparkline_in_7d && data.market_data.sparkline_in_7d.price) {
          var sparklinePrices = data.market_data.sparkline_in_7d.price;
          var sparklineColor = change7d > 0 ? '#22c55e' : '#ef4444';
          var svg = generateSparklineSVG(sparklinePrices, 200, 60, sparklineColor);
          document.getElementById('sol-sparkline').innerHTML = svg;
        } else {
          document.getElementById('sol-sparkline').innerHTML = '<img src="https://www.coingecko.com/coins/4128/sparkline.svg" style="width: 100%; height: 60px; object-fit: cover;" alt="Solana 7D Chart">';
        }
      }
      // Fetch TVL
      try {
        var tvlResponse = await fetch('https://api.llama.fi/tvl/solana');
        var tvlData = await tvlResponse.json();
        var tvlFormatted = tvlData > 1e12 ? '$' + (tvlData / 1e12).toFixed(1) + 'T' : '$' + (tvlData / 1e9).toFixed(0) + 'B';
        document.getElementById('sol-tvl').textContent = tvlFormatted;
      } catch (e) {
        document.getElementById('sol-tvl').textContent = 'Data unavailable';
      }
    } catch (error) {
      console.error('Error:', error);
      document.getElementById('sol-price').textContent = 'Data unavailable';
      document.getElementById('sol-change').textContent = 'Data unavailable';
    }
  }
  document.addEventListener('DOMContentLoaded', updateSolanaData);
